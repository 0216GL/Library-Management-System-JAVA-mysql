-- =============================================
-- 图书馆管理系统数据库
-- 适配中国手机号（11位）
-- 字符集：utf8mb4（支持中文）
-- 图书使用ISBN作为唯一标识符
-- =============================================

-- 创建新数据库
CREATE DATABASE library
CHARACTER SET utf8mb4
COLLATE utf8mb4_general_ci;

USE library;

-- =============================================
-- 1. 人员表（基表）
-- =============================================
CREATE TABLE PERSON
(
    ID INT NOT NULL PRIMARY KEY,
    PNAME VARCHAR(30) NOT NULL,
    PASSWORD VARCHAR(10) NOT NULL,
    ADDRESS VARCHAR(30) NOT NULL,
    PHONE_NO VARCHAR(15) NOT NULL
);

-- =============================================
-- 2. 图书表
-- =============================================
CREATE TABLE BOOK
(
    ISBN VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '国际标准书号',
    TITLE VARCHAR(30) NOT NULL,
    AUTHOR VARCHAR(20) NOT NULL,
    SUBJECT VARCHAR(20) NOT NULL,
    IS_ISSUED BOOLEAN NOT NULL
);

-- =============================================
-- 3. 员工表
-- =============================================
CREATE TABLE STAFF
(
    S_ID INT NOT NULL PRIMARY KEY,
    TYPE VARCHAR(10) NOT NULL,
    SALARY DOUBLE,
    FOREIGN KEY (S_ID) REFERENCES PERSON(ID)
);

-- =============================================
-- 4. 办事员表
-- =============================================
CREATE TABLE CLERK
(
    C_ID INT NOT NULL PRIMARY KEY,
    DESK_NO INT NOT NULL,
    FOREIGN KEY (C_ID) REFERENCES STAFF(S_ID)
);

-- =============================================
-- 5. 图书管理员表
-- =============================================
CREATE TABLE LIBRARIAN
(
    L_ID INT NOT NULL PRIMARY KEY,
    OFFICE_NO INT NOT NULL,
    FOREIGN KEY (L_ID) REFERENCES STAFF(S_ID)
);

-- =============================================
-- 6. 借阅者表
-- =============================================
CREATE TABLE BORROWER
(
    B_ID INT NOT NULL PRIMARY KEY,
    FOREIGN KEY (B_ID) REFERENCES PERSON(ID)
);

-- =============================================
-- 7. 借阅记录表
-- =============================================
CREATE TABLE LOAN
(
    L_ID INT NOT NULL PRIMARY KEY,
    BORROWER INT NOT NULL,
    BOOK VARCHAR(20) NOT NULL COMMENT '图书ISBN',
    ISSUER INT NOT NULL,
    ISS_DATE TIMESTAMP NOT NULL,
    RECEIVER INT,
    RET_DATE TIMESTAMP,
    FINE_PAID BOOLEAN,
    FOREIGN KEY (BORROWER) REFERENCES BORROWER(B_ID),
    FOREIGN KEY (ISSUER) REFERENCES STAFF(S_ID),
    FOREIGN KEY (RECEIVER) REFERENCES STAFF(S_ID),
    FOREIGN KEY (BOOK) REFERENCES BOOK(ISBN)
);

-- =============================================
-- 8. 借出的书（中间表）
-- =============================================
CREATE TABLE BORROWED_BOOK
(
    BOOK VARCHAR(20) NOT NULL PRIMARY KEY COMMENT '图书ISBN',
    BORROWER INT NOT NULL,
    FOREIGN KEY (BORROWER) REFERENCES BORROWER(B_ID),
    FOREIGN KEY (BOOK) REFERENCES BOOK(ISBN)
);

-- =============================================
-- 9. 预约表
-- =============================================
CREATE TABLE ON_HOLD_BOOK
(
    REQ_ID INT NOT NULL PRIMARY KEY,
    BOOK VARCHAR(20) NOT NULL COMMENT '图书ISBN',
    BORROWER INT NOT NULL,
    REQ_DATE DATE NOT NULL,
    FOREIGN KEY (BORROWER) REFERENCES BORROWER(B_ID),
    FOREIGN KEY (BOOK) REFERENCES BOOK(ISBN)
);
